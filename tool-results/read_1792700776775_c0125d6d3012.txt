    10→ *  1· Las 28 fichas de la baraja CAEN DEL CIELO con tumbo 3D, sombra de contacto

    15→ * Se puede saltar tocando la pantalla y respeta prefers-reduced-motion.

    17→
    18→interface ScatterTile {
    19→  a: number
    20→  b: number
    21→  x: number // % del ancho del fieltro
    22→  y: number // % del alto del fieltro
    23→  rot: number
    24→  horizontal: boolean
    25→  delay: number
    26→  fall: number // duración de la caída (s)
    27→  z: number
    28→  size: 'xs' | 'sm'
    29→  /** deriva horizontal durante la caída (px) */
    30→  fx: number
    31→  /** altura de entrada: arranca por encima de la pantalla (px) */
    32→  fy: number
    33→  /** tumbo 3D en grados */
    34→  flipX: number
    35→  flipY: number
    36→  spin: number
    37→}
    38→
    39→const STORM_MS = 1150 // la mesa se oscurece: viene el rayo
    40→const BOLT_MS = 1900 // el rayo cae
    41→const CRACK_MS = 2150 // la mesa quiebra
    42→const READY_MS = 3200 // aparece JUGAR
    43→
    44→/** Reparto aleatorio de las 28 fichas sobre el fieltro (con hueco libre en el centro) */
    45→function makeScatter(w: number, h: number): ScatterTile[] {
    46→  const deck: Array<[number, number]> = []
    47→  for (let a = 0; a <= 6; a++) for (let b = a; b <= 6; b++) deck.push([a, b])
    48→  for (let i = deck.length - 1; i > 0; i--) {
    49→    const j = Math.floor(Math.random() * (i + 1))
    50→    ;[deck[i], deck[j]] = [deck[j], deck[i]]
    51→  }
    52→
    53→  const size: 'xs' | 'sm' = w >= 560 ? 'sm' : 'xs'
    54→  const minDist = (size === 'sm' ? 72 : 56) * 0.62
    55→  const placed: Array<{ x: number; y: number }> = []
    56→
    57→  return deck.map(([a, b], i) => {
    58→    let x = 50
    59→    let y = 50
    60→    for (let t = 0; t < 60; t++) {
    61→      x = 8 + Math.random() * 84
    62→      y = 9 + Math.random() * 82
    63→      // deja el centro despejado para el botón de Jugar
    64→      if (Math.abs(x - 50) < 17 && Math.abs(y - 52) < 25) continue
    65→      const px = (x / 100) * w
    66→      const py = (y / 100) * h
    67→      let ok = true
    68→      for (const p of placed) {
    69→        const dx = px - (p.x / 100) * w
    70→        const dy = py - (p.y / 100) * h
    71→        if (dx * dx + dy * dy < minDist * minDist) {
    72→          ok = false
    73→          break
    74→        }
    75→      }
    76→      if (ok) break
    77→    }
    78→    placed.push({ x, y })
    79→
    80→    return {
    81→      a,
    82→      b,
    83→      x,
    84→      y,
    85→      rot: -42 + Math.random() * 84,
    86→      horizontal: Math.random() < 0.5,
    87→      delay: i * 0.032 + Math.random() * 0.05,
    88→      fall: 0.78 + Math.random() * 0.28,
    89→      z: Math.floor(Math.random() * 9),
    90→      size,
    91→      fx: -90 + Math.random() * 180,
    92→      fy: -(h + 160 + Math.random() * 280),
    93→      flipX: (Math.random() < 0.5 ? -1 : 1) * (260 + Math.random() * 400),
    94→      flipY: (Math.random() < 0.5 ? -1 : 1) * (Math.random() * 170),
    95→      spin: Math.random() * 190 - 95,
    96→    }
    97→  })
    98→}

   100→/** Brasas ambientales flotando sobre la mesa rota (fase ready) */
   101→function makeEmbers(): Array<{ x: number; y: number; s: number; d: number; dur: number; drift: number; ember: boolean }> {
   102→  return Array.from({ length: 14 }, () => ({
   103→    x: 6 + Math.random() * 88,
   104→    y: 30 + Math.random() * 62,
   105→    s: 2 + Math.random() * 2.4,
   106→    d: Math.random() * 2.2,
   107→    dur: 2.3 + Math.random() * 1.7,
   108→    drift: -14 + Math.random() * 28,
   109→    ember: Math.random() < 0.55,
   110→  }))
   111→}

   143→type IntroPhase = 'pre' | 'throwing' | 'bolt' | 'crack' | 'ready'
   144→
   145→export function TableIntro({
   146→  children,
   147→  onPhaseChange,
   148→}: {
   149→  children: ReactNode
   150→  onPhaseChange?: (phase: IntroPhase) => void
   151→}) {
   152→  const rootRef = useRef<HTMLDivElement>(null)
   153→  const timersRef = useRef<Array<ReturnType<typeof setTimeout>>>([])
   154→  const [phase, setPhase] = useState<IntroPhase>('pre')
   155→  const [storm, setStorm] = useState(false)
   156→  const [tiles, setTiles] = useState<ScatterTile[]>([])
   157→  const [reduced, setReduced] = useState(false)
   158→  const gradientId = useId().replace(/[^a-zA-Z0-9_-]/g, '')
   159→
   160→  const onPhaseRef = useRef(onPhaseChange)
   161→  onPhaseRef.current = onPhaseChange
   162→  /** cambia de fase y avisa a la mesa (la bandera se revela con el rayo) */
   163→  const go = (p: IntroPhase) => {
   164→    setPhase(p)
   165→    onPhaseRef.current?.(p)
   166→  }
   167→
   168→  useEffect(() => {
   169→    const isReduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches
   170→    if (isReduced) {
   171→      setReduced(true)
   172→      go('ready')
   173→      return
   174→    }
   175→    const rect = rootRef.current?.getBoundingClientRect()
   176→    const scatter = makeScatter(rect?.width ?? 700, rect?.height ?? 420)
   177→    setTiles(scatter)
   178→    go('throwing')
   179→
   180→    const timers = timersRef.current
   181→    // golpes sordos de cada ficha al aterrizar (~60% de su caída)
   182→    for (const t of scatter) {
   183→      timers.push(setTimeout(() => playSfx('thud'), (t.delay + t.fall * 0.62) * 1000))
   184→    }
   185→    timers.push(setTimeout(() => setStorm(true), STORM_MS))
   186→    timers.push(
   187→      setTimeout(() => {
   188→        playSfx('thunder')
   189→        go('bolt')
   190→      }, BOLT_MS),
   191→    )
   192→    timers.push(setTimeout(() => go('crack'), CRACK_MS))
   193→    timers.push(setTimeout(() => go('ready'), READY_MS))
   194→    return () => timers.forEach(clearTimeout)
   195→  }, [])
   196→
   197→  // Un toque durante la intro la salta (el click no llega al botón: aún no existe)
   198→  const skip = () => {
   199→    if (phase === 'ready' || phase === 'pre') return
   200→    timersRef.current.forEach(clearTimeout)
   201→    go('ready')
   202→  }
   203→
   204→  const impact = phase === 'bolt' || phase === 'crack'
   205→  const embers = useMemo(() => (phase === 'ready' && !reduced ? makeEmbers() : []), [phase, reduced])
   206→
   207→  return (
   208→    <div
   209→      ref={rootRef}
   210→      data-testid="table-intro"
   211→      data-phase={phase}
   212→      onClickCapture={phase === 'ready' ? undefined : skip}
   213→      className="absolute inset-0"
   214→    >
   215→      {/* ===== capa de atmósfera: oscuridad de tormenta + viñeta de cine ===== */}
   216→      <motion.div
   217→        className="pointer-events-none absolute inset-0 bg-black"
   218→        aria-hidden
   219→        initial={{ opacity: 0 }}
   220→        animate={{
   221→          opacity:
   222→            phase === 'bolt'
   223→              ? 0.32
   224→              : phase === 'crack'
   225→                ? 0.14
   226→                : phase === 'ready'
   227→                  ? 0.12
   228→                  : storm
   229→                    ? 0.46
   230→                    : 0,
   231→        }}
   232→        transition={{ duration: phase === 'bolt' ? 0.12 : 0.55, ease: 'easeOut' }}
   233→      />
   234→      <motion.div
   235→        className="pointer-events-none absolute inset-0 bg-[radial-gradient(ellipse_at_center,transparent_42%,rgba(0,0,0,0.62)_100%)]"
   236→        aria-hidden
   237→        initial={{ opacity: 0 }}
   238→        animate={{ opacity: phase === 'ready' ? 0.4 : storm || impact ? 0.8 : 0 }}
   239→        transition={{ duration: 0.6 }}
   240→      />
   241→
   242→      {/* ===== cámara: todo lo físico se sacude con el impacto ===== */}
   243→      <motion.div
   244→        className="absolute inset-0"
   245→        style={{ perspective: 1100 }}
   246→        animate={
   247→          phase === 'bolt'
   248→            ? {
   249→                x: [0, -10, 9, -6, 4, -2, 0],
   250→                y: [0, 6, -6, 4, -2, 1, 0],
   251→                scale: [1, 1.03, 1.014, 1],
   252→              }
   253→            : { x: 0, y: 0, scale: 1 }
   254→        }
   255→        transition={{ duration: 0.55, ease: 'easeOut' }}

   257→        {/* ---- capa 1: sombras de contacto (bajo cada ficha que cae) ---- */}
