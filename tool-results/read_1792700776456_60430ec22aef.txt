   106→interface DropZones {
   107→  left: DOMRect[]
   108→  right: DOMRect[]

   114→const ZONE_PAD = 44

   116→const AUTO_PASS_DELAY = 1250

   118→/** Mide las zonas de suelta de la mesa: extremo izquierdo y extremo derecho de la cadena */
   119→function collectDropZones(chainStarted: boolean): DropZones | null {
   120→  if (typeof document === 'undefined') return null
   121→  const zones: DropZones = { left: [], right: [] }
   122→  if (!chainStarted) {
   123→    const empty = document.querySelector<HTMLElement>('[data-end="empty"]')
   124→    if (empty) {
   125→      const r = empty.getBoundingClientRect()
   126→      zones.left.push(r)
   127→      zones.right.push(r)
   128→    }
   129→    return zones.left.length > 0 ? zones : null
   130→  }
   131→  const leftEl = document.querySelector<HTMLElement>('[data-end="left"], [data-end="both"]')
   132→  const rightEl = document.querySelector<HTMLElement>('[data-end="right"], [data-end="both"]')
   133→  const lr = leftEl?.getBoundingClientRect() ?? null
   134→  const rr = rightEl?.getBoundingClientRect() ?? null
   135→  if (lr && rr && Math.abs(lr.left - rr.left) < 1 && Math.abs(lr.width - rr.width) < 1) {
   136→    // Con una sola ficha en la mesa ella misma es ambos extremos: mitad izquierda / mitad derecha
   137→    zones.left.push(new DOMRect(lr.left, lr.top, lr.width / 2, lr.height))
   138→    zones.right.push(new DOMRect(lr.left + lr.width / 2, lr.top, lr.width / 2, lr.height))
   139→    zones.left.push(new DOMRect(lr.left - 80, lr.top, 80, lr.height))
   140→    zones.right.push(new DOMRect(lr.right, lr.top, 80, lr.height))
   141→  } else {
   142→    if (lr) {
   143→      zones.left.push(lr)
   144→      // prolongación hacia fuera del extremo: ahí es donde apoyarías la ficha nueva
   145→      zones.left.push(new DOMRect(lr.left - 90, lr.top, 90, lr.height))
   146→    }
   147→    if (rr) {
   148→      zones.right.push(rr)
   149→      zones.right.push(new DOMRect(rr.right, rr.top, 90, rr.height))
   150→    }
   151→  }
   152→  return zones.left.length > 0 || zones.right.length > 0 ? zones : null
   153→}
   154→
   155→/** ¿Está el puntero sobre alguna zona de suelta válida? Devuelve el lado más cercano */
   156→function hitTestSide(zones: DropZones | null, x: number, y: number, sides: Side[]): Side | null {
   157→  if (!zones) return null
   158→  let best: Side | null = null
   159→  let bestDist = Infinity
   160→  for (const side of ['left', 'right'] as const) {
   161→    if (!sides.includes(side)) continue
   162→    for (const r of zones[side]) {
   163→      if (
   164→        x < r.left - ZONE_PAD ||
   165→        x > r.right + ZONE_PAD ||
   166→        y < r.top - ZONE_PAD ||
   167→        y > r.bottom + ZONE_PAD
   168→      ) {
   169→        continue
   170→      }
   171→      const dist = Math.abs(x - (r.left + r.width / 2))
   172→      if (dist < bestDist) {
   173→        bestDist = dist
   174→        best = side
   175→      }
   176→    }
   177→  }
   178→  return best
   179→}
