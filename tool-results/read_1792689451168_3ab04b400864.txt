    50→interface DropZones {
    51→  left: DOMRect[]
    52→  right: DOMRect[]
    53→}

    55→/** Movimiento mínimo (px) para distinguir un arrastre de un simple toque */
    56→const DRAG_THRESHOLD = 6
    57→/** Amplitud de cada zona de suelta alrededor de su extremo (px) */
    58→const ZONE_PAD = 44
    59→/** Retardo del paso automático cuando no tienes jugada (ms) — como en la mesa real */
    60→const AUTO_PASS_DELAY = 1250

    62→/** Mide las zonas de suelta de la mesa: extremo izquierdo y extremo derecho de la cadena */
    63→function collectDropZones(chainStarted: boolean): DropZones | null {
    64→  if (typeof document === 'undefined') return null
    65→  const zones: DropZones = { left: [], right: [] }
    66→  if (!chainStarted) {
    67→    const empty = document.querySelector<HTMLElement>('[data-end="empty"]')
    68→    if (empty) {
    69→      const r = empty.getBoundingClientRect()
    70→      zones.left.push(r)
    71→      zones.right.push(r)
    72→    }
    73→    return zones.left.length > 0 ? zones : null
    74→  }
    75→  const leftEl = document.querySelector<HTMLElement>('[data-end="left"], [data-end="both"]')
    76→  const rightEl = document.querySelector<HTMLElement>('[data-end="right"], [data-end="both"]')
    77→  const lr = leftEl?.getBoundingClientRect() ?? null
    78→  const rr = rightEl?.getBoundingClientRect() ?? null
    79→  if (lr && rr && Math.abs(lr.left - rr.left) < 1 && Math.abs(lr.width - rr.width) < 1) {
    80→    // Con una sola ficha en la mesa ella misma es ambos extremos: mitad izquierda / mitad derecha
    81→    zones.left.push(new DOMRect(lr.left, lr.top, lr.width / 2, lr.height))
    82→    zones.right.push(new DOMRect(lr.left + lr.width / 2, lr.top, lr.width / 2, lr.height))
    83→    zones.left.push(new DOMRect(lr.left - 80, lr.top, 80, lr.height))
    84→    zones.right.push(new DOMRect(lr.right, lr.top, 80, lr.height))
    85→  } else {
    86→    if (lr) {
    87→      zones.left.push(lr)
    88→      // prolongación hacia fuera del extremo: ahí es donde apoyarías la ficha nueva
    89→      zones.left.push(new DOMRect(lr.left - 90, lr.top, 90, lr.height))
    90→    }
    91→    if (rr) {
    92→      zones.right.push(rr)
    93→      zones.right.push(new DOMRect(rr.right, rr.top, 90, rr.height))
    94→    }
    95→  }
    96→  return zones.left.length > 0 || zones.right.length > 0 ? zones : null
    97→}

    99→/** ¿Está el puntero sobre alguna zona de suelta válida? Devuelve el lado más cercano */
   100→function hitTestSide(zones: DropZones | null, x: number, y: number, sides: Side[]): Side | null {
   101→  if (!zones) return null
   102→  let best: Side | null = null
   103→  let bestDist = Infinity
   104→  for (const side of ['left', 'right'] as const) {
   105→    if (!sides.includes(side)) continue
   106→    for (const r of zones[side]) {
   107→      if (
   108→        x < r.left - ZONE_PAD ||
   109→        x > r.right + ZONE_PAD ||
   110→        y < r.top - ZONE_PAD ||
   111→        y > r.bottom + ZONE_PAD
   112→      ) {
   113→        continue
   114→      }
   115→      const dist = Math.abs(x - (r.left + r.width / 2))
   116→      if (dist < bestDist) {
   117→        bestDist = dist
   118→        best = side
   119→      }
   120→    }
   121→  }
   122→  return best
   123→}
